3) 🔒 Security Rules (เวอร์ชันแก้ไข — ปลอดภัยจาก DoS)

rules_version = '2';
service cloud.firestore {
  match /databases/{db}/documents {
    function signedIn() { return request.auth != null; }
    function isMember() {
      return signedIn() &&
        exists(/databases/$(db)/documents/users/$(request.auth.uid));
    }
    function isMaster() {
      return isMember() &&
        get(/databases/$(db)/documents/users/$(request.auth.uid)).data.role == 'master';
    }
    function bootDone() {
      return exists(/databases/$(db)/documents/meta/bootstrap);
    }

    match /meta/{id} {
      allow read: if signedIn();
      // 🔒 FIX v3.5: ต้องมี master user ใน batch เดียวกัน
      allow create: if signedIn()
        && id == 'bootstrap'
        && !bootDone()
        && getAfter(/databases/$(db)/documents/users/$(request.auth.uid)).data.role == 'master';
    }

    match /users/{uid} {
      allow read: if signedIn() && (request.auth.uid == uid || isMember());
      allow create: if request.auth.uid == uid
        && request.resource.data.badge.upper() + '@roster.local' == request.auth.token.email
        && ((request.resource.data.role == 'user'
              && exists(/databases/$(db)/documents/employees/$(request.resource.data.badge)))
            || (request.resource.data.role == 'master' && !bootDone()));
      allow update, delete: if isMaster();
    }

    match /employees/{badge} {
      allow read: if isMember();
      allow write: if isMaster();
    }

    match /bookings/{id} {
      allow read: if isMember();
      allow create: if isMember() && request.resource.data.createdBy == request.auth.uid;
      allow update, delete: if isMaster() || (isMember() && resource.data.createdBy == request.auth.uid);
    }
  }
}
